# Minimal "component agent" - paste into Kibana > Dev Tools > Console.
# Component table: component-registry, one document per component:
#   { "component": "MDSG", "aliases": ["MDSG", "mdsg-prod"], "indices": ["logs-generic.otel-mdsg", ...] }
# Names/aliases can be in any case: the lookup is case-insensitive (verified on 9.4.6).

# ---- Tool: look up a component's indices (use PUT .../registry.component_indices to update an existing one,
#      body without "id" and "type")
POST kbn:/api/agent_builder/tools
{
  "id": "registry.component_indices",
  "type": "esql",
  "description": "Returns the indices that hold a component's logs, metrics and traces. Call this FIRST for any question about a component. Matching is case-insensitive on the component name and its aliases.",
  "configuration": {
    "query": "FROM component-registry | WHERE TO_LOWER(component) == TO_LOWER(?component) OR MV_CONTAINS(TO_LOWER(aliases), TO_LOWER(?component)) | KEEP component, aliases, indices | LIMIT 1",
    "params": {
      "component": { "type": "string", "description": "Component name or alias exactly as the user wrote it (any case)" }
    }
  }
}

# ---- Agent (for an existing agent use: PUT kbn:/api/agent_builder/agents/component-agent  with the same body minus "id")
POST kbn:/api/agent_builder/agents
{
  "id": "component-agent",
  "name": "Component Agent",
  "description": "Answers questions about a component using only that component's indices (logs, traces and metrics).",
  "configuration": {
    "instructions": "You answer questions about one component at a time, using ONLY that component's own indices.\n1) Call registry.component_indices with the component name as the user wrote it. If nothing is found, say so and stop.\n2) The tool returns ALL indices of the component: logs-* (logs), traces-* (traces) and metrics-* (metrics). Check EVERY returned index, not only logs. For each index, first call platform.core.get_index_mapping to learn its field names, then run at least one query on it with platform.core.execute_esql:\n   - logs: errors/warnings over time (severity_text, severity_number) and the most frequent error messages (body.text)\n   - traces: number of spans, failed spans (status.code == \"Error\") and p95 duration by span name\n   - metrics: whether data is still arriving (latest @timestamp) and which metrics changed\n3) ES|QL rules: write index names in FROM exactly as returned and WITHOUT backticks (correct: FROM logs-generic.otel-mdsg ; wrong: FROM `logs-generic.otel-mdsg`). Backticks are only for field names. Always filter on @timestamp for the time window and add a LIMIT. Only use field names that exist in the mapping.\n4) If a query fails, read the error, fix the query and retry once. If an index really does not exist, name it in the answer.\n5) Never query an index the tool did not return, and never search the whole cluster.\n6) Answer per signal (logs / traces / metrics) with numbers, hosts and times, and list the indices you used.",
    "tools": [
      {
        "tool_ids": [
          "registry.component_indices",
          "platform.core.get_index_mapping",
          "platform.core.execute_esql"
        ]
      }
    ]
  }
}

# ---- Check the tool directly (should return MDSG's indices)
POST kbn:/api/agent_builder/tools/_execute
{ "tool_id": "registry.component_indices", "tool_params": { "component": "mdsg" } }
